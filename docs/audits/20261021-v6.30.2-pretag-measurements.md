## D#511: timing at the user-prompt hook's input cap

`shapes.mjs` builds each crafted shape so the UserPromptSubmit stdin JSON stays within 256 KiB;
`e2e2.mjs <SP> <headTree> <newTree> 5` runs `hook.mjs user-prompt` on each shape, the two trees
alternating per trial, each process recording its own CPU time through `cpu-preload.mjs`.
`headTree` = `git archive bcbe1f51` (its scrubber is v6.30.1's: secret-scrub.mjs is unchanged from 5443e879 to c4adcc05~1); `newTree` = the repo at c4adcc05.
Run 2026-10-21 ~06:35 with load average 13-17 from another session, so absolute times are inflated
and the multiples of benign (median CPU / median benign CPU, same run) are the reading:

(gitignored) spec tasks/specs/remaining-after-6301-20261021.md; it is not in the repo.

Equality with v6.30.1's scrubber: the archived D#500 generator (docs/audits/20261020-v6.30.1-pretag-measurements.md,
`d500-eq.mjs`) on seeds 1, 2, 3, 51, 52, 71, 72, 101, 102 x 30,000 = 270,000 inputs, 0 differences;
`tagfuzz.mjs` seeds 1-3 x 100,000, 0 differences. `unitfuzz.mjs` checks closingTagLength against the
regex it replaced (about 6,500 tag matches per seed, 0 differences) and the call-site blank class

### shapes.mjs
```js
// Worst known scrubber shapes, sized so the UserPromptSubmit stdin JSON stays within the 256 KiB cap.
export const CAP = 256 * 1024;
const CHAIN = Array.from({ length: 40 }, (_, i) => (i % 2 ? 'secret: abcdefgh' : 'token: abcdefgh')).join(' ');
const hdr = (pre, br = '\n') => Array.from({ length: 16 }, (_, i) => `${pre}X-h${String.fromCharCode(65 + i)}: -----BEGIN RSA PRIVATE KEY-----${br}`).join('');
const prose = (n) => 'the quick brown fox jumps over a lazy dog '.repeat(Math.ceil(n / 42) + 1).slice(0, n);
const B = '-----BEGIN RSA PRIVATE KEY-----';
export const gens = {
  benign: (n) => CHAIN + ' ' + prose(n),
  d500: (n) => CHAIN + ` ${B}\n` + hdr('') + '\n'.repeat(n),
  spaces: (n) => CHAIN + ` ${B}\n` + hdr('') + ' \n'.repeat(n / 2),
  quotes: (n) => CHAIN + ` ${B}\n` + hdr('') + '"\n'.repeat(n / 2),
  escaped: (n) => CHAIN + ` ${B}\\n` + hdr('').replace(/\n/g, '\\n') + '\\n'.repeat(n / 2),
  beginBlankWord: (n) => `-----BEGIN PRIVATE KEY-----\n\nword\n`.repeat(Math.ceil(n / 34)),
  tags: (n) => CHAIN + ` ${B}\n` + hdr('') + '</a>\n'.repeat(n / 5),
  liveQuote: (n) => CHAIN + `\n"${B}\n` + hdr('"') + '"\n'.repeat(n / 2),
  liveTag: (n) => CHAIN + `\n</${B}\n` + Array.from({ length: 16 }, (_, i) => `</X-h${String.fromCharCode(65 + i)}:${B}\n`).join('') + Array.from({ length: Math.floor(n / 9) }, (_, i) => `</t${i % 997}>\n`).join(''),
  quoteComma: (n) => CHAIN + ` ${B}\n` + hdr('') + '"",\n'.repeat(n / 4),
  headerCap: (n) => CHAIN + ` ${B}\n` + Array.from({ length: Math.floor(n / 40) }, (_, i) => `X-h${i % 26}: ${B}\n`).join(''),
  depthsNoRun: (n) => {
    let s = CHAIN;
    for (let k = 0; k < 12; k++) { const br = '\\'.repeat(2 ** k) + 'n'; s += ' ' + B + br + hdr('', br); }
    return s + ' ' + prose(n);
  },
};
export const payload = (prompt) => JSON.stringify({ session_id: 'd511-session', hook_event_name: 'UserPromptSubmit', cwd: process.env.D511_CWD || '/tmp', prompt });
// Largest n whose payload fits the cap.
export function fit(gen) {
  let lo = 1000, hi = 600000;
  while (hi - lo > 200) { const mid = Math.floor((lo + hi) / 2); if (Buffer.byteLength(payload(gen(mid))) <= CAP) lo = mid; else hi = mid; }
  return gen(lo);
}
```

### e2e2.mjs
```js
// user-prompt hook end to end, HEAD and new alternating per trial, wall and CPU time per process.
import { spawnSync } from 'node:child_process';
import { mkdtempSync, mkdirSync, rmSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { gens, fit, payload } from './shapes.mjs';
const [SP, headRoot, newRoot, roundsArg] = process.argv.slice(2);
const arms = { head: headRoot, new: newRoot };
const root = mkdtempSync(join(SP, 'e2e2-'));
const log = join(root, 'cpu.jsonl');
const env = { ...process.env, HOME: join(root, 'home'), CLAUDE_PROJECT_DIR: join(root, 'proj'), CLAUDE_CODE_PATH: join(root, 'no-claude'), MEM_NO_AUTO_ADOPT: '1', D511_CPU_LOG: log };
delete env.ANTHROPIC_API_KEY; delete env.OPENROUTER_API_KEY; delete env.CLAUDE_MEM_HOOK_RUNNING;
for (const d of ['home', 'proj', 'data-head', 'data-new']) mkdirSync(join(root, d), { recursive: true });
process.env.D511_CWD = join(root, 'proj');
const texts = Object.fromEntries(Object.entries(gens).map(([k, g]) => [k, payload(fit(g))]));
const run = (arm, input) => {
  writeFileSync(log, '');
  const s = performance.now();
  const r = spawnSync(process.execPath, ['--import', join(SP, 'cpu-preload.mjs'), arms[arm] + '/hook.mjs', 'user-prompt'], { input, env: { ...env, CLAUDE_MEM_DIR: join(root, 'data-' + arm) }, cwd: join(root, 'proj'), encoding: 'utf8', timeout: 30000 });
  const wall = performance.now() - s;
  const cpu = JSON.parse(readFileSync(log, 'utf8').trim().split('\n').pop()).cpuMs;
  if (r.status !== 0) throw new Error(`${arm} exit ${r.status} ${r.stderr}`);
  return { wall, cpu };
};
for (const a of ['head', 'new']) run(a, texts.benign);
const res = {};
for (let round = 0; round < Number(roundsArg); round++) for (const [k, p] of Object.entries(texts)) for (const a of round % 2 ? ['new', 'head'] : ['head', 'new']) { ((res[k] ||= {})[a] ||= []).push(run(a, p)); }
const med = (xs) => { const v = [...xs].sort((x, y) => x - y); return v[Math.floor(v.length / 2)]; };
const bw = (a) => med(res.benign[a].map((r) => r.wall)), bc = (a) => med(res.benign[a].map((r) => r.cpu));
console.log('shape            HEAD wall  cpu  xbenign |  new wall  cpu  xbenign | new/HEAD cpu');
for (const [k, v] of Object.entries(res)) {
  const f = (a) => `${med(v[a].map((r) => r.wall)).toFixed(0).padStart(6)} ${med(v[a].map((r) => r.cpu)).toFixed(0).padStart(5)} ${(med(v[a].map((r) => r.cpu)) / bc(a)).toFixed(1).padStart(6)}x`;
  console.log(`${k.padEnd(15)} ${f('head')} | ${f('new')} | ${(med(v.new.map((r) => r.cpu)) / med(v.head.map((r) => r.cpu))).toFixed(2)}`);
}
rmSync(root, { recursive: true, force: true });
```

### unitfuzz.mjs
```js
// Unit equivalence on the shipped text: closingTagLength vs the regex it replaced, and the call-site
// blank class vs lineCore under random prefix shapes.
import { readFileSync } from 'node:fs';
const src = readFileSync('/home/ai/dev/claude-mem-lite/secret-scrub.mjs', 'utf8');
const grab = (start, end) => { const i = src.indexOf(start); return src.slice(i, src.indexOf(end, i)); };
const body = grab('const isTagNameCode', '/** The judged part of line') + grab('/** A regex for the line prefix', 'const isTagNameCode') + grab('/** The judged part of line', 'function classify(core)');
const f = new Function(`${body}; return { closingTagLength, lineCore, prefixShape };`)();
let seed = Number(process.argv[2] || 1);
const rnd = () => { seed = (Math.imul(seed, 1103515245) + 12345) & 0x7fffffff; return seed / 0x7fffffff; };
const pick = (s) => s[Math.floor(rnd() * s.length)];
const ALPHA = ['<', '/', '>', 'a', 'Z', 'q', '0', '_', ':', '.', '-', ' ', '"', "'", '`', 'x', '\t', '+', ',', ')', '\\', 'n'];
let tagCases = 0, tagDiffs = 0, tagHits = 0;
for (let it = 0; it < 300000; it++) {
  let line = '';
  const mode = rnd();
  if (mode < 0.5) { const nl = Math.floor(rnd() * 46); line = pick(['', ' ', '"', 'x']) + '</' + pick(['a', 'Z', '0', '_', '']) + Array.from({ length: nl }, () => pick(['a', 'b', '0', '_', ':', '.', '-', '/', 'c', 'd', 'e', 'f', 'g', 'h'])).join('') + '>' + pick(['', ' ', ',', ')']); }
  else { const n = Math.floor(rnd() * 14); for (let i = 0; i < n; i++) line += pick(ALPHA); }
  let a = Math.floor(rnd() * (line.length + 1)); let q = a + Math.floor(rnd() * (line.length - a + 1));
  if (mode < 0.5 && rnd() < 0.5) { q = line.lastIndexOf('>') + 1; a = rnd() < 0.5 ? 0 : Math.floor(rnd() * q); }
  if (!(q > a)) continue;
  tagCases++;
  const m = /<\/[A-Za-z][\w:.-]{0,40}>$/.exec(line.slice(a, q));
  const want = m ? m[0].length : 0; const got = f.closingTagLength(line, a, q);
  if (want) tagHits++;
  if (want !== got) { if (tagDiffs++ < 3) console.log('TAG DIFF', JSON.stringify({ line, a, q, want, got })); }
}
console.log(`closingTagLength: ${tagCases} cases, ${tagHits} with a tag, ${tagDiffs} differences`);
// Call-site class: [ \t]* (one of " ' `)? [ \t]*  => lineCore core === '' for any shape.
const isBlankClass = (t) => /^[ \t]*["'`]?[ \t]*$/.test(t);
let blankCases = 0, blankDiffs = 0, shapesSeen = 0;
for (let it = 0; it < 300000; it++) {
  const pfx = Array.from({ length: Math.floor(rnd() * 6) }, () => pick([' ', '\t', '"', "'", '`', '-', '+', 'X', ':', '1', '#', '<', '/'])).join('');
  const shape = f.prefixShape(pfx);
  if (shape) shapesSeen++;
  const n = Math.floor(rnd() * 6); let line = ''; for (let i = 0; i < n; i++) line += pick([' ', '\t', '"', "'", '`', ' ', '-', '+', 'X']);
  if (!isBlankClass(line)) continue;
  blankCases++;
  const text = 'ab\n' + line + '\ncd'; const s = 3, e = 3 + line.length;
  const r = f.lineCore(text, s, e, shape);
  if (r.core !== '') { if (blankDiffs++ < 3) console.log('BLANK DIFF', JSON.stringify({ pfx, line, core: r.core })); }
}
console.log(`blank class: ${blankCases} lines in the class, ${shapesSeen} with a shape, ${blankDiffs} where lineCore kept a core`);
```

### tagfuzz.mjs
```js
// Focused equality: BEGINs under prefixes that can start a blank-ish line, over runs of tag/quote lines.
const [oldDir, newDir, itersArg, seedArg] = process.argv.slice(2);
const { scrubSecrets: o } = await import(`${oldDir}/secret-scrub.mjs`);
const { scrubSecrets: n } = await import(`${newDir}/secret-scrub.mjs`);
let seed = Number(seedArg);
const rnd = () => { seed = (Math.imul(seed, 1103515245) + 12345) & 0x7fffffff; return seed / 0x7fffffff; };
const pick = (a) => a[Math.floor(rnd() * a.length)];
const PREFIX = ['', ' ', '<', '</', '"', '"<', "'", '`', 'X-h:', '1', '"</', '<a>', '  </', '-', '+', '> ', '"X'];
const LINE = () => {
  const ws = () => pick(['', ' ', '\t', '  ']);
  const q = pick(['', '', '"', "'", '`']);
  const name = pick(['a', 'b-c', 'X:y', 'p.q', 'a1', '', '1a', 'a/b', 'a'.repeat(40), 'a'.repeat(41)]);
  const tag = pick([`</${name}>`, `<${name}>`, `</${name}`, '</>', `</${name}>x`, `x</${name}>`, '']);
  const tail = pick(['', '', ',', ' ', ')', '+']);
  return ws() + q + pick(['', '', ' ']) + tag + tail + ws();
};
const BODY = () => pick(['MIIEabcdefghijklmnopqrstuvwxyz0123456789ABCD', 'abcdefghijklmnop', 'Zm9vYmFyYmF6cXV4', '-----END RSA PRIVATE KEY-----', 'X-hB: v', 'word']);
let diffs = 0, ex = [];
for (let i = 0; i < Number(itersArg); i++) {
  const p = pick(PREFIX);
  let t = pick(['', 'token: abcdefgh secret: abcdefgh ']) + p + '-----BEGIN RSA PRIVATE KEY-----\n';
  const lines = Math.floor(rnd() * 8);
  for (let j = 0; j < lines; j++) t += (rnd() < 0.5 ? p : pick(PREFIX)) + LINE() + '\n';
  if (rnd() < 0.7) t += (rnd() < 0.5 ? p : '') + BODY() + '\n';
  if (rnd() < 0.3) t += (rnd() < 0.5 ? p : '') + BODY() + '\n';
  const a = o(t), b = n(t);
  if (a !== b) { diffs++; if (ex.length < 3) ex.push({ t, a, b }); }
}
console.log(JSON.stringify({ iters: Number(itersArg), seed: Number(seedArg), diffs }));
for (const e of ex) console.log(JSON.stringify(e));
```
